def encrypt(text, rails):
    if rails == 1:
        return text

    fence = [""] * rails
    row = 0
    direction = 1

    for ch in text:
        fence[row] += ch

        if row == 0:
            direction = 1
        elif row == rails - 1:
            direction = -1

        row += direction

    return ''.join(fence)


def decrypt(cipher, rails):
    if rails == 1:
        return cipher

    pattern = []
    row = 0
    direction = 1

    for i in range(len(cipher)):
        pattern.append(row)

        if row == 0:
            direction = 1
        elif row == rails - 1:
            direction = -1

        row += direction

    counts = [pattern.count(i) for i in range(rails)]

    fence = []
    index = 0

    for count in counts:
        fence.append(list(cipher[index:index + count]))
        index += count

    result = ""
    positions = [0] * rails

    for r in pattern:
        result += fence[r][positions[r]]
        positions[r] += 1

    return result


text = input("Enter message: ")
rails = int(input("Enter number of rails: "))

encrypted = encrypt(text, rails)
print("Encrypted:", encrypted)

decrypted = decrypt(encrypted, rails)
print("Decrypted:", decrypted)
